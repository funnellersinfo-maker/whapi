    35→  | { kind: "intro" }

    44→const stepVariants = {
    45→  enter: (dir: number) => ({ opacity: 0, x: 26 * dir }),

    51→  invalid_name: "Escribe tu nombre para continuar.",

    54→  invalid_session: "Selecciona la fecha y hora de tu sesión.",
    55→  invalid_visitor: "No pudimos identificarte. Recarga la página e inténtalo de nuevo.",
    56→  invalid_quiz: "Faltan respuestas del diagnóstico. Vuelve al inicio del formulario.",
    57→  invalid_booking: "Faltan datos de la reserva. Inténtalo de nuevo.",
    58→  server_error: "No pudimos guardar tu sesión. Inténtalo de nuevo.",
    59→};

    61→function progressOf(step: Step): number {
    62→  switch (step.kind) {
    63→    case "intro":
    64→      return 0;
    65→    case "question":
    66→      return (step.index + 1) * 10;
    67→    case "filter":
    68→      return 90;
    69→    default:
    70→      return 100;
    71→  }
    72→}

    74→function OptionCard({
    75→  label,
    76→  selected,
    77→  locked,
    78→  onClick,
    79→}: {
    80→  label: string;
    81→  selected: boolean;
    82→  locked: boolean;
    83→  onClick: () => void;
    84→}) {
    85→  return (
    86→    <motion.button
    87→      type="button"
    88→      onClick={onClick}
    89→      disabled={locked}
    90→      whileTap={{ scale: 0.98 }}
    91→      className={cn(
    92→        "flex w-full items-center justify-between gap-3 rounded-2xl border px-4 py-[15px] text-left transition-all duration-200",
    93→        selected
    94→          ? "border-wa/70 bg-wa/[0.09] shadow-[0_0_28px_-8px_rgba(0,230,118,0.5)]"
    95→          : "border-white/10 bg-white/[0.025] hover:border-white/25 hover:bg-white/[0.05]"
    96→      )}
    97→    >
    98→      <span
    99→        className={cn(
   100→          "text-[14.5px] font-medium leading-snug",
   101→          selected ? "text-ink" : "text-ink/85"
   102→        )}
   103→      >
   104→        {label}
   105→      </span>
   106→      <span
   107→        className={cn(
   108→          "flex h-6 w-6 shrink-0 items-center justify-center rounded-full border transition-all duration-200",
   109→          selected ? "border-wa bg-wa text-[#03150c]" : "border-white/15 text-transparent"
   110→        )}
   111→      >
   112→        <motion.span
   113→          initial={false}
   114→          animate={selected ? { scale: 1 } : { scale: 0.3 }}
   115→          transition={{ type: "spring", stiffness: 500, damping: 26 }}
   116→        >
   117→          <Check className="h-3.5 w-3.5" strokeWidth={3.5} />
   118→        </motion.span>
   119→      </span>
   120→    </motion.button>
   121→  );
   122→}

   135→export function Quiz() {
   136→  const [step, setStep] = useState<Step>({ kind: "intro" });
   137→  const [dir, setDir] = useState(1);
   138→  const [answers, setAnswers] = useState<Partial<QuizData>>({});
   139→  const [pending, setPending] = useState<string | null>(null);
   140→  const [multi, setMulti] = useState<string[]>([]);
   141→  const [businessText, setBusinessText] = useState("");
   142→  const [toolText, setToolText] = useState("");
   143→  const [sessionDate, setSessionDate] = useState("");
   144→  const [sessionTime, setSessionTime] = useState("");
   145→  const [form, setForm] = useState({ name: "", whatsapp: "", email: "" });
   146→  const [formError, setFormError] = useState("");
   147→  const [submitting, setSubmitting] = useState(false);
   148→  const [waSent, setWaSent] = useState(false);
   149→
   150→  const visitorRef = useRef("");
   151→  const timers = useRef<number[]>([]);
   152→  const cardRef = useRef<HTMLDivElement>(null);
   153→  const calendarViewed = useRef(false);
   154→
   155→  const days = useMemo(() => getUpcomingDays(21), []);
   156→
   157→  useEffect(() => {
   158→    initPixel();
   159→    trackStandard("ViewContent", { content_name: "landing_quiz_view" });
   160→    try {
   161→      const stored = window.localStorage.getItem("wa-sys-visitor");
   162→      const id =
   163→        stored ??
   164→        (typeof crypto !== "undefined" && crypto.randomUUID
   165→          ? crypto.randomUUID()
   166→          : `v-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`);
   167→      if (!stored) window.localStorage.setItem("wa-sys-visitor", id);
   168→      visitorRef.current = id;
   169→    } catch {
   170→      visitorRef.current = `v-${Date.now()}`;
   171→    }
   172→    return () => {
   173→      timers.current.forEach((t) => window.clearTimeout(t));
   174→    };
   175→  }, []);
   176→
   177→  const saveProgress = useCallback(
   178→    (quiz: Partial<QuizData>, action: "progress" | "disqualified" = "progress") => {
   179→      const visitorId = visitorRef.current;
   180→      if (!visitorId) return;
   181→      void fetch("/api/lead", {
   182→        method: "POST",
   183→        headers: { "Content-Type": "application/json" },
   184→        body: JSON.stringify({ visitorId, action, quiz }),
   185→      }).catch(() => undefined);
   186→    },
   187→    []
   188→  );
   189→
   190→  const scrollToCard = useCallback(() => {
   191→    requestAnimationFrame(() => {
   192→      cardRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   193→    });
   194→  }, []);
   195→
   196→  const schedule = useCallback((fn: () => void, ms: number) => {
   197→    const t = window.setTimeout(fn, ms);
   198→    timers.current.push(t);
   199→  }, []);
   200→
   201→  const go = useCallback(
   202→    (next: Step) => {
   203→      setDir(1);
   204→      setPending(null);
   205→      setStep(next);
   206→      scrollToCard();
   207→    },
   208→    [scrollToCard]
   209→  );
   210→
   211→  const back = useCallback(() => {
   212→    setDir(-1);
   213→    setPending(null);
   214→    if (step.kind === "question" && step.index > 0) {
   215→      setStep({ kind: "question", index: step.index - 1 });
   216→    } else if (step.kind === "filter") {
   217→      setStep({ kind: "question", index: QUESTIONS.length - 1 });
   218→    } else if (step.kind === "form") {
   219→      setStep({ kind: "calendar" });
   220→    }
   221→    scrollToCard();
   222→  }, [step, scrollToCard]);
   223→
   224→  const advanceFrom = useCallback(
   225→    (index: number) => {
   226→      if (index + 1 >= QUESTIONS.length) {
   227→        trackCustom("quiz_completed");
   228→        setStep({ kind: "filter" });
   229→      } else {
   230→        setStep({ kind: "question", index: index + 1 });
   231→      }
   232→      setDir(1);
   233→      setPending(null);
   234→      scrollToCard();
   235→    },
   236→    [scrollToCard]
   237→  );
   238→
   239→  const onSingle = useCallback(
   240→    (field: string, value: string) => {
   241→      if (pending || step.kind !== "question") return;
   242→      setPending(value);
   243→      const updated = { ...answers, [field]: value } as Partial<QuizData>;
   244→      setAnswers(updated);
   245→      saveProgress(updated);
   246→
   247→      if (field === "automationPrev") {
   248→        if (value !== "No, sería la primera vez") return; // reveal optional input
   249→      } else if (field === "decisionMaker") {
   250→        if (value !== "Yo tomo la decisión") return; // reveal note
   251→      }
   252→      schedule(() => advanceFrom(step.index), 480);
   253→    },
   254→    [advanceFrom, answers, pending, saveProgress, schedule, step]
   255→  );
   256→
   257→  const onContinueQuestion = useCallback(() => {
   258→    if (step.kind !== "question") return;
   259→    const q = QUESTIONS[step.index];
   260→    if (q.id === "businessType") {
   261→      const value = businessText.trim();
   262→      if (value.length < 2) return;
   263→      const updated = { ...answers, businessType: value };
   264→      setAnswers(updated);
   265→      saveProgress(updated);
   266→    } else if (q.id === "trafficSources") {
   267→      if (multi.length === 0) return;
   268→      const updated = { ...answers, trafficSources: multi };
   269→      setAnswers(updated);
   270→      saveProgress(updated);
   271→    } else if (q.id === "automationPrev") {
   272→      const updated = { ...answers, automationTool: toolText.trim() || undefined };
   273→      setAnswers(updated);
   274→      saveProgress(updated);
   275→    }
   276→    advanceFrom(step.index);
   277→  }, [advanceFrom, answers, businessText, multi, saveProgress, step, toolText]);
   278→
   279→  const toggleMulti = useCallback((opt: string) => {
   280→    setMulti((prev) => {
   281→      if (opt === NO_TRAFFIC_OPTION) {
   282→        return prev.includes(opt) ? [] : [opt];
   283→      }
   284→      const base = prev.filter((x) => x !== NO_TRAFFIC_OPTION);
   285→      return base.includes(opt) ? base.filter((x) => x !== opt) : [...base, opt];
   286→    });
   287→  }, []);
   288→
   289→  const onFilter = useCallback(
   290→    (value: BudgetConfirmed) => {
   291→      if (pending) return;
   292→      setPending(value);
   293→      const updated = { ...answers, budgetConfirmed: value };
   294→      setAnswers(updated);
   295→      if (value === "NO_POR_AHORA") {
   296→        saveProgress(updated, "disqualified");
   297→        trackCustom("disqualified_lead");
   298→        try {
   299→          window.dispatchEvent(new Event("landing:quiz-completed"));
   300→        } catch {
   301→          /* noop */
   302→        }
   303→        schedule(() => go({ kind: "disqualified" }), 650);
   304→      } else {
   305→        saveProgress(updated);
   306→        trackCustom("qualified_lead");
   307→        trackStandard("CompleteRegistration");
   308→        schedule(() => go({ kind: "analyzing" }), 650);
   309→      }
   310→    },
   311→    [answers, go, pending, saveProgress, schedule]
   312→  );
   313→
   314→  useEffect(() => {
   315→    if (step.kind === "analyzing") {
   316→      const t = window.setTimeout(() => {
   317→        setStep({ kind: "calendar" });
   318→        scrollToCard();
   319→      }, 2750);
   320→      timers.current.push(t);
   321→      return () => window.clearTimeout(t);
   322→    }
   323→    if (step.kind === "calendar" && !calendarViewed.current) {
   324→      calendarViewed.current = true;
   325→      trackCustom("calendar_viewed");
   326→    }
   327→  }, [step, scrollToCard]);
   328→
   329→  const formatWa = (raw: string) => {
   330→    const d = raw.replace(/\D/g, "").slice(0, 10);
   331→    const parts = [d.slice(0, 3), d.slice(3, 6), d.slice(6, 10)].filter(Boolean);
   332→    return parts.join(" ");
   333→  };
   334→
   335→  const onBook = useCallback(async () => {
   336→    setFormError("");
   337→    const name = form.name.trim();
   338→    const digits = form.whatsapp.replace(/\D/g, "");
   339→    const email = form.email.trim();
   340→    if (name.length < 2) {
   341→      setFormError(ERROR_MESSAGES.invalid_name);
   342→      return;
   343→    }
   344→    if (!/^3\d{9}$/.test(digits)) {
   345→      setFormError(ERROR_MESSAGES.invalid_whatsapp);
   346→      return;
   347→    }
   348→    if (email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   349→      setFormError(ERROR_MESSAGES.invalid_email);
   350→      return;
   351→    }
   352→    if (!sessionDate || !sessionTime) {
   353→      setFormError(ERROR_MESSAGES.invalid_session);
   354→      return;
   355→    }
   356→    setSubmitting(true);
   357→    try {
   358→      const res = await fetch("/api/lead", {
   359→        method: "POST",
   360→        headers: { "Content-Type": "application/json" },
   361→        body: JSON.stringify({
   362→          visitorId: visitorRef.current,
   363→          action: "booked",
   364→          quiz: answers,
   365→          booking: { name, whatsapp: digits, email, sessionDate, sessionTime },
   366→        }),
   367→      });
   368→      const data = (await res.json().catch(() => null)) as
   369→        | { ok?: boolean; waUrl?: string; error?: string }
   370→        | null;
   371→      if (!res.ok || !data?.ok || !data.waUrl) {
   372→        setFormError(
   373→          ERROR_MESSAGES[data.error ?? "server_error"] ??
   374→            ERROR_MESSAGES.server_error
   375→        );
   376→        setSubmitting(false);
   377→        return;
   378→      }
   379→      trackCustom("appointment_booked");
   380→      trackStandard("Lead");
   381→      trackStandard("Schedule");
   382→      try {
   383→        window.dispatchEvent(new Event("landing:booking-complete"));
   384→      } catch {
   385→        /* noop */
   386→      }
   387→      setStep({ kind: "success", waUrl: data.waUrl });
   388→      scrollToCard();
   389→      try {
   390→        window.open(data.waUrl, "_blank", "noopener,noreferrer");
   391→      } catch {
   392→        /* popup blocked: user has the button */
   393→      }
   394→    } catch {
   395→      setFormError("No pudimos guardar tu sesión. Revisa tu conexión e inténtalo de nuevo.");
   396→    } finally {
   397→      setSubmitting(false);
   398→    }
   399→  }, [answers, form, scrollToCard, sessionDate, sessionTime]);
   400→
   401→  const onConfirmWhatsapp = useCallback(() => {
   402→    if (step.kind !== "success") return;
   403→    if (!waSent) {
   404→      setWaSent(true);
   405→      trackCustom("whatsapp_sent");
   406→      trackStandard("Contact");
   407→    }
   408→    window.open(step.waUrl, "_blank", "noopener,noreferrer");
   409→  }, [step, waSent]);
   410→
   411→  const pct = progressOf(step);
   412→
   413→  const wrap = (key: string, children: ReactNode) => (
   414→    <motion.div
   415→      key={key}
   416→      custom={dir}
   417→      variants={stepVariants}
   418→      initial="enter"
   419→      animate="center"
   420→      exit="exit"
   421→      transition={{ duration: 0.34, ease: [0.22, 1, 0.36, 1] }}
   422→    >
   423→      {children}
   424→    </motion.div>
   425→  );
   426→
   427→  const suggestions = useMemo(() => {
   428→    const q = (businessText ?? "").trim().toLowerCase();
   429→    const norm = (s: string) =>
   430→      s
   431→        .toLowerCase()
   432→        .normalize("NFD")
   433→        .replace(/[\u0300-\u036f]/g, "");
   434→    if (!q) return BUSINESS_SUGGESTIONS.slice(0, 8);
   435→    return BUSINESS_SUGGESTIONS.filter((s) => norm(s).includes(norm(q))).slice(0, 8);
   436→  }, [businessText]);
   437→
   438→  const renderStep = (): ReactNode => {
   439→    switch (step.kind) {
   440→      case "intro":
   441→        return wrap(
   442→          "intro",
   443→          <div className="flex min-h-[420px] flex-col items-center justify-center text-center">
   444→            <div className="flex h-14 w-14 items-center justify-center rounded-2xl border border-wa/25 bg-wa/10">
   445→              <Clock className="h-6 w-6 text-wa" />
   446→            </div>
   447→            <h3 className="font-display mt-5 text-[24px] font-bold uppercase leading-tight text-ink">
   448→              Diagnóstico de tu negocio
   449→            </h3>
   450→            <p className="mt-3 max-w-xs text-[14.5px] leading-relaxed text-dim">
   451→              8 preguntas rápidas para saber si tu negocio es candidato a este
   452→              sistema.
   453→            </p>
   454→            <div className="mt-5 flex flex-wrap items-center justify-center gap-2 text-[11.5px] text-dim">
   455→              <span className="inline-flex items-center gap-1.5 rounded-full border border-white/10 bg-white/[0.03] px-3 py-1.5">
   456→                <Clock className="h-3.5 w-3.5" /> Menos de 60 segundos
   457→              </span>
   458→              <span className="inline-flex items-center gap-1.5 rounded-full border border-white/10 bg-white/[0.03] px-3 py-1.5">
   459→                <ShieldCheck className="h-3.5 w-3.5" /> Sin compromiso
   460→              </span>
   461→            </div>
   462→            <div className="mt-8 w-full">
   463→              <CtaButton
   464→                onClick={() => {
   465→                  trackCustom("quiz_started");
   466→                  go({ kind: "question", index: 0 });
   467→                }}
   468→              >
   469→                Comenzar diagnóstico
   470→              </CtaButton>
   471→            </div>
   472→          </div>
   473→        );
   474→
   475→      case "question": {
   476→        const q = QUESTIONS[step.index];
   477→        return wrap(
   478→          `q-${step.index}`,
   479→          <div>
   480→            <div className="flex items-center justify-between">
   481→              <button
   482→                type="button"
   483→                onClick={back}
   484→                className="inline-flex items-center gap-1 text-[11px] font-semibold uppercase tracking-wider text-dim transition-colors hover:text-ink"
   485→              >
   486→                <ChevronLeft className="h-3.5 w-3.5" />
   487→                Atrás
   488→              </button>
   489→              <span className="text-[10px] font-bold uppercase tracking-[0.22em] text-dim">
   490→                Pregunta {step.index + 1} de {QUESTIONS.length}
   491→              </span>
   492→            </div>
   493→            <h3 className="font-display mt-4 text-[20px] font-bold leading-[1.18] text-ink">
   494→              {q.title}
   495→            </h3>
   496→            {q.helper ? (
   497→              <p className="mt-2 text-[13px] leading-relaxed text-dim">{q.helper}</p>
   498→            ) : null}
   499→
   500→            {q.kind === "text" ? (
   501→              <div className="mt-5 space-y-4">
   502→                <input
   503→                  value={businessText}
   504→                  onChange={(e) => setBusinessText(e.target.value)}
   505→                  placeholder="Ej: tienda de ropa, barbería, consultora…"
   506→                  maxLength={60}
   507→                  className="h-12 w-full rounded-xl border border-white/12 bg-white/[0.04] px-4 text-[15px] text-ink outline-none transition-colors placeholder:text-dim/60 focus:border-wa/60"
   508→                />
   509→                <div className="flex flex-wrap gap-1.5">
   510→                  {suggestions.map((s) => (
   511→                    <button
   512→                      key={s}
   513→                      type="button"
   514→                      onClick={() => setBusinessText(s)}
   515→                      className={cn(
   516→                        "rounded-full border px-3 py-1.5 text-[12px] transition-colors",
   517→                        businessText === s
   518→                          ? "border-wa/60 bg-wa/10 text-ink"
   519→                          : "border-white/10 bg-white/[0.03] text-ink/75 hover:border-wa/40 hover:text-ink"
   520→                      )}
   521→                    >
   522→                      {s}
   523→                    </button>
   524→                  ))}
   525→                </div>
   526→                <CtaButton
   527→                  size="md"
   528→                  dimmed={businessText.trim().length < 2}
   529→                  onClick={onContinueQuestion}
   530→                >
   531→                  Continuar
   532→                </CtaButton>
   533→              </div>
   534→            ) : null}
   535→
   536→            {q.kind === "single" ? (
   537→              <div className="mt-5 space-y-2.5">
   538→                {q.options?.map((opt) => (
   539→                  <OptionCard
   540→                    key={opt}
   541→                    label={opt}
   542→                    selected={pending === opt || answers[q.id] === opt}
   543→                    locked={pending !== null}
   544→                    onClick={() => onSingle(q.id, opt)}
   545→                  />
   546→                ))}
   547→              </div>
   548→            ) : null}
   549→
   550→            {q.kind === "multi" ? (
   551→              <div>
   552→                <div className="mt-5 space-y-2.5">
   553→                  {q.options?.map((opt) => (
   554→                    <OptionCard
   555→                      key={opt}
   556→                      label={opt}
   557→                      selected={multi.includes(opt)}
   558→                      locked={false}
   559→                      onClick={() => toggleMulti(opt)}
   560→                    />
   561→                  ))}
   562→                </div>
   563→                <div className="mt-5">
   564→                  <CtaButton
   565→                    size="md"
   566→                    dimmed={multi.length === 0}
   567→                    onClick={onContinueQuestion}
   568→                  >
   569→                    {multi.length === 0 ? "Selecciona al menos una" : "Continuar"}
   570→                  </CtaButton>
   571→                </div>
   572→              </div>
   573→            ) : null}
   574→
